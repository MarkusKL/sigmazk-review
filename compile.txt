         (R' : Stat → Wit' → Prop) (Err : Forb → Prop)

         (∀ (h : Stat) (w : Wit), R h w → R' h (toW w))
         → (∀ (h : Stat) (w : Wit'), R' h w → case (R h) Err (fromW w))
           → sigma Chal R' Err → sigma Chal R Err
